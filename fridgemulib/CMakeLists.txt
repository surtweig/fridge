cmake_minimum_required(VERSION 3.15)
project(fridgemulib)
set(CMAKE_CXX_STANDARD 14)
include_directories(../include)
add_executable(fridgemulib main.cpp fridgemulib.c fridgemulib.h posit.c posit.h)

include(CTest)
if(BUILD_TESTING)
    if(NOT TARGET falc_core)
        add_subdirectory(../falc falc EXCLUDE_FROM_ALL)
    endif()
    add_executable(rom_library_test tests/rom_library.cpp fridgemulib.c)
    target_include_directories(rom_library_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(rom_library_test PRIVATE falc_core)
    add_test(NAME rom_library COMMAND rom_library_test)

    add_executable(rom_bootloader_test tests/rom_bootloader.cpp fridgemulib.c)
    target_include_directories(rom_bootloader_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(rom_bootloader_test PRIVATE
        FRIDGE_BOOTLOADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../fridge-boot/BootLoader/")
    target_link_libraries(rom_bootloader_test PRIVATE falc_core)
    add_test(NAME rom_bootloader COMMAND rom_bootloader_test)

    set(FRIDGE_TEST_ARTIFACT_DIR "${CMAKE_CURRENT_BINARY_DIR}/artifacts" CACHE PATH
        "Directory for emulator test screenshots")
    file(MAKE_DIRECTORY "${FRIDGE_TEST_ARTIFACT_DIR}")
    foreach(mode IN ITEMS 210x160 240x180)
        add_executable(video_${mode} tests/video_modes.cpp fridgemulib.c)
        target_include_directories(video_${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions(video_${mode} PRIVATE
            FRIDGE_TEST_ARTIFACT_DIR="${FRIDGE_TEST_ARTIFACT_DIR}")
        if(mode STREQUAL "240x180")
            target_compile_definitions(video_${mode} PRIVATE FRIDGE_VIDEO_240X180=1)
        endif()
        add_test(NAME video_${mode} COMMAND video_${mode})
    endforeach()
    foreach(mode IN ITEMS 210x160 240x160 240x180)
        add_executable(rle_${mode} tests/rle_lines.cpp fridgemulib.c)
        target_include_directories(rle_${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions(rle_${mode} PRIVATE
            FRIDGE_TEST_ARTIFACT_DIR="${FRIDGE_TEST_ARTIFACT_DIR}")
        if(mode STREQUAL "240x160")
            target_compile_definitions(rle_${mode} PRIVATE FRIDGE_VIDEO_240X160=1)
        elseif(mode STREQUAL "240x180")
            target_compile_definitions(rle_${mode} PRIVATE FRIDGE_VIDEO_240X180=1)
        endif()
        add_test(NAME rle_${mode} COMMAND rle_${mode})
    endforeach()
endif()
