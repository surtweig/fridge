cmake_minimum_required(VERSION 3.15)
project(fridgemulib)
set(CMAKE_CXX_STANDARD 14)
include_directories(../include)
add_executable(fridgemulib main.cpp fridgemulib.c fridgemulib.h posit.c posit.h)

include(CTest)
if(BUILD_TESTING)
    foreach(mode IN ITEMS 210x160 240x180)
        add_executable(video_${mode} tests/video_modes.cpp fridgemulib.c)
        target_include_directories(video_${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        if(mode STREQUAL "240x180")
            target_compile_definitions(video_${mode} PRIVATE FRIDGE_VIDEO_240X180=1)
        endif()
        add_test(NAME video_${mode} COMMAND video_${mode})
    endforeach()
endif()
