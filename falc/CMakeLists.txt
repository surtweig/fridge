cmake_minimum_required(VERSION 3.15)
project(falc)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_EXE_LINKER_FLAGS "-static")
add_library(falc_core STATIC FridgeAssemblyLanguageCompiler.cpp FridgeAssemblyLanguageCompiler.h)
target_include_directories(falc_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../include")
get_filename_component(FALC_STD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/x2al_std" ABSOLUTE)
target_compile_definitions(falc_core PRIVATE FALC_STD_PATH="${FALC_STD_DIR}/")

add_executable(falc main.cpp)
target_link_libraries(falc PRIVATE falc_core)
